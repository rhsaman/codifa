مقایسه رندرینگ بلوک کد با کامنت فارسی/انگلیسی

❌ بدون dir="ltr" (وضعیت فعلی شما)

tapped from parent dir="rtl" → مشکل bidi
// HashMap نقش را بازی می‌کند
const map = new Map();
tapped from parent dir="rtl" → کلمات جابه‌جا
# کاربر user.id را ست کرد
user_id = 1

✅ با dir="ltr" + unicode-bidi: plaintext

dir="ltr" + unicode-bidi: plaintext → درست
// HashMap نقش را بازی می‌کند
const map = new Map();
dir="ltr" + unicode-bidi: plaintext → درست
# کاربر user.id را ست کرد
user_id = 1
کامنت فقط فارسی → همچنان RTL
// این تابع کاربر را احراز هویت می‌کند
function auth(user) {}
کامنت فقط انگلیسی → همچنان LTR
// authenticate the user
function auth(user) {}

نکته: کافیه یکی از دو dir="ltr" یا unicode-bidi: plaintext رو برداری تا مشکل برگرده.
